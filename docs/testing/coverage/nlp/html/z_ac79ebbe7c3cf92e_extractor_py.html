1"""Resolve DATE spans to timezone-aware datetimes. 

2 

3Relative expressions resolve against the message timestamp in the user's 

4timezone. Date-only phrases get a default due time (end of business). Vague 

5phrases resolve to a best guess but are flagged ambiguous, so they do not 

6affect priority until the user confirms them. 

7""" 

8 

9import calendar 

10import re 

11from dataclasses import dataclass 

12from datetime import datetime, time, timedelta 

13from zoneinfo import ZoneInfo 

14 

15import dateparser 

16 

17from .taglish_cleaner import gloss_time_expression 

18 

19DEFAULT_DUE_TIME = time(17, 0) 

20 

21 

22@dataclass(frozen=True) 

23class ResolvedDate: 

24 value: datetime | None 

25 ambiguous: bool

28def resolve_date(phrase: str, *, received_at: datetime, timezone: str) -> ResolvedDate: 

29 zone = ZoneInfo(timezone) 

30 base = received_at.astimezone(zone) 

31 text = gloss_time_expression(phrase) 

32 text, part_of_day = _split_part_of_day(text) 

33 explicit_time = _explicit_time(text) 

34 date_text = _TIME_RE.sub("", text).strip() 

35 

36 special = _resolve_special(date_text, base) 

37 if special is not None: 

38 day, ambiguous = special 

39 else: 

40 parsed = _parse_with_dateparser(date_text, base, timezone) if date_text else base 

41 if parsed is None: 

42 return ResolvedDate(None, True) 

43 day = parsed.astimezone(zone) 

44 ambiguous = _is_numeric_day_month_ambiguous(date_text) 

45 

46 due_time = explicit_time or part_of_day or DEFAULT_DUE_TIME 

47 resolved = datetime.combine(day.date(), due_time, tzinfo=zone) 

48 return ResolvedDate(resolved, ambiguous)

51_PART_OF_DAY_TIMES = { 

52 "morning": time(9, 0), 

53 "noon": time(12, 0), 

54 "afternoon": time(15, 0), 

55 "evening": time(19, 0), 

56 "night": time(20, 0), 

57 "midnight": time(23, 59), 

58} 

59_TIME_RE = re.compile( 

60 r"(?:\s*(?:at\s+|@\s*|alas[\s-]+)?(?P<hour>\d{1,2})(?::(?P<minute>\d{2}))?\s*(?P<meridiem>am|pm|a\.m\.|p\.m\.|nn|n\.n\.))(?![a-z])", 

61 re.IGNORECASE, 

62) 

63_WEEKDAY_INDEX = {name: index for index, name in enumerate(calendar.day_name)} 

64_WEEKDAY_ABBREVIATIONS = { 

65 "mon": "monday", "tue": "tuesday", "tues": "tuesday", "wed": "wednesday", 

66 "thu": "thursday", "thur": "thursday", "thurs": "thursday", "fri": "friday", 

67}

70def _split_part_of_day(text: str) -> tuple[str, time | None]: 

71 if re.search(r"\btonight\b", text): 

72 return re.sub(r"\btonight\b", "today", text), time(20, 0) 

73 if re.search(r"\blater today\b", text): 

74 return text, None 

75 for word, value in _PART_OF_DAY_TIMES.items(): 

76 if re.search(rf"\b(?:this\s+)?{word}\b", text): 

77 return re.sub(rf"\s*\b(?:this\s+)?{word}\b", "", text).strip() or "today", value 

78 return text, None

81def _explicit_time(text: str) -> time | None: 

82 match = _TIME_RE.search(text) 

83 if not match: 

84 return None 

85 hour = int(match.group("hour")) 

86 minute = int(match.group("minute") or 0) 

87 meridiem = match.group("meridiem").casefold().replace(".", "") 

88 if meridiem == "nn": 

89 return time(12, minute) 

90 if hour > 12: 

91 return None 

92 if meridiem == "pm" and hour != 12: 

93 hour += 12 

94 if meridiem == "am" and hour == 12: 

95 hour = 0 

96 return time(hour, minute)

99def _resolve_special(text: str, base: datetime) -> tuple[datetime, bool] | None: 

100 """Expressions dateparser handles poorly or that need an ambiguity flag.""" 

101 text = re.sub(r"^(?:by|on|before|until|the)\s+", "", text.strip(" .,")) 

102 today = base.replace(hour=0, minute=0, second=0, microsecond=0) 

103 

104 if text in {"today", ""}: 

105 return today, False 

106 if text == "later today": 

107 return today, True 

108 if text in {"tomorrow", "the next day"}: 

109 return today + timedelta(days=1), False 

110 if text in {"the day after tomorrow", "day after tomorrow"}: 

111 return today + timedelta(days=2), False 

112 if text in {"end of day", "end of the day", "eod", "cob"}: 

113 return today, False 

114 if text in {"end of week", "end of the week", "eow", "this week"}: 

115 return today + timedelta(days=(4 - today.weekday()) % 7), text == "this week" 

116 if text in {"end of month", "end of the month", "this month"}: 

117 last = calendar.monthrange(today.year, today.month)[1] 

118 return today.replace(day=last), text == "this month" 

119 if text == "next week": 

120 return today + timedelta(days=7 - today.weekday()), True 

121 if text == "next month": 

122 first_next = (today.replace(day=1) + timedelta(days=32)).replace(day=1) 

123 return first_next, True 

124 

125 weekday = re.fullmatch(r"(?:(?P<modifier>next|this|coming)\s+)?(?P<day>[a-z]+)\.?", text) 

126 if weekday: 

127 name = _WEEKDAY_ABBREVIATIONS.get(weekday.group("day"), weekday.group("day")) 

128 if name.capitalize() in _WEEKDAY_INDEX: 

129 target = _WEEKDAY_INDEX[name.capitalize()] 

130 days_ahead = (target - today.weekday()) % 7 

131 modifier = weekday.group("modifier") 

132 # "Friday" sent on a Friday, or "next Friday", can mean two different weeks. 

133 ambiguous = days_ahead == 0 or modifier == "next" 

134 if days_ahead == 0: 

135 days_ahead = 7 

136 return today + timedelta(days=days_ahead), ambiguous 

137 return None

140def _parse_with_dateparser(text: str, base: datetime, timezone: str) -> datetime | None: 

141 return dateparser.parse( 

142 text, 

143 languages=["en"], 

144 settings={ 

145 "RELATIVE_BASE": base.replace(tzinfo=None), 

146 "PREFER_DATES_FROM": "future", 

147 "TIMEZONE": timezone, 

148 "TO_TIMEZONE": timezone, 

149 "RETURN_AS_TIMEZONE_AWARE": True, 

150 "DATE_ORDER": "MDY", 

151 }, 

152 )

155def _is_numeric_day_month_ambiguous(text: str) -> bool: 

156 match = re.fullmatch(r"(\d{1,2})/(\d{1,2})(?:/\d{2,4})?", text) 

157 if not match: 

158 return False 

159 first, second = int(match.group(1)), int(match.group(2)) 

160 return first <= 12 and second <= 12 and first != second
