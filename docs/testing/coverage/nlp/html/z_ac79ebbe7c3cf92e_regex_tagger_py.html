1"""Bounded regex baseline for TASK and DATE spans. 

2 

3Stand-in for the Bi-LSTM-CRF tagger (Phase 4) so the pipeline works end to 

4end, and later the comparison baseline and fallback. It deliberately covers 

5only a fixed list of English, Filipino and Taglish patterns. 

6""" 

7 

8import re 

9from dataclasses import dataclass 

10from typing import Literal 

11 

12SpanLabel = Literal["TASK", "DATE"] 

13 

14MAX_TASK_CHARS = 120 

15 

16 

17@dataclass(frozen=True) 

18class RawSpan: 

19 label: SpanLabel 

20 text: str 

21 start: int 

22 end: int 

23 

24 

25def tag(text: str) -> list[RawSpan]: 

26 dates = _find_dates(text) 

27 tasks = _find_tasks(text, dates) 

28 return sorted([*dates, *tasks], key=lambda span: (span.start, span.label))

31def has_deadline_marker_before(text: str, start: int) -> bool: 

32 window = text[max(0, start - 24) : start].casefold() 

33 return bool(_DEADLINE_MARKER_RE.search(window))

36_WEEKDAYS_EN = r"monday|tuesday|wednesday|thursday|friday|saturday|sunday|(?:mon|tues|tue|wed|thurs|thur|thu|fri)\.?(?=\s|$|[,;:!?)])" 

37_WEEKDAYS_TL = "lunes|martes|miyerkules|miyerkoles|huwebes|biyernes|byernes|sabado|linggo" 

38_MONTHS = ( 

39 "january|february|march|april|may|june|july|august|september|october|november|december|" 

40 "jan|feb|mar|apr|jun|jul|aug|sep|sept|oct|nov|dec|" 

41 "enero|pebrero|marso|abril|mayo|hunyo|hulyo|agosto|setyembre|oktubre|nobyembre|disyembre" 

42) 

43_PART_OF_DAY = r"(?:\s+(?:ng\s+)?(?:morning|afternoon|evening|night|noon|umaga|tanghali|hapon|gabi))?" 

44_TIME = r"(?:\s*(?:at\s+|@\s*|ng\s+|alas[\s-]+)?\d{1,2}(?::\d{2})?\s*(?:am|pm|a\.m\.|p\.m\.|nn|n\.n\.)(?![a-z]))" 

45 

46_DATE_PATTERNS = [ 

47 # Filipino multi-word expressions first so they win over their parts. 

48 r"(?:sa\s+)?susunod\s+na\s+(?:linggo|buwan)", 

49 r"katapusan\s+ng\s+buwan", 

50 r"(?:sa\s+)?(?:sa)?makalawa", 

51 r"mamayang\s+(?:gabi|hapon|umaga|tanghali)", 

52 r"ngayong\s+(?:araw|linggo|buwan)", 

53 r"(?:bukas|mamaya|ngayon|kinabukasan)" + _PART_OF_DAY, 

54 rf"(?:sa\s+)?(?:{_WEEKDAYS_TL})", 

55 # English relative expressions. 

56 r"(?:the\s+)?day\s+after\s+tomorrow", 

57 r"(?:tomorrow|today|tonight)" + _PART_OF_DAY, 

58 r"end\s+of\s+(?:the\s+)?(?:day|week|month)", 

59 r"\b(?:eod|eow|cob)\b", 

60 r"(?:next|this)\s+(?:week|month)", 

61 rf"(?:next\s+|this\s+|coming\s+)?(?:{_WEEKDAYS_EN})" + _PART_OF_DAY, 

62 # Absolute dates. 

63 rf"(?:{_MONTHS})\.?\s+\d{{1,2}}(?:st|nd|rd|th)?(?:,?\s*\d{{4}})?", 

64 rf"\d{{1,2}}(?:st|nd|rd|th)?\s+(?:of\s+)?(?:{_MONTHS})(?:,?\s*\d{{4}})?", 

65 r"\d{4}-\d{2}-\d{2}", 

66 r"\d{1,2}/\d{1,2}(?:/\d{2,4})?", 

67] 

68_DATE_RE = re.compile( 

69 r"(?<![\w-])(?:" + "|".join(_DATE_PATTERNS) + r")(?:" + _TIME + r")?(?![\w-])", 

70 re.IGNORECASE, 

71)

73_TASK_TRIGGERS = [ 

74 r"paki(?:-|\s)?[a-z]+", 

75 r"pakisuyo(?:ng)?", 

76 r"(?:please|pls|plz|kindly)\s+(?!(?:find|see|note|be\s+advised|disregard|ignore)\b)[a-z]+", 

77 r"(?:can|could|would)\s+you(?:\s+please)?\s+[a-z]+", 

78 r"(?:make\s+sure|be\s+sure|remember|don't\s+forget|do\s+not\s+forget)\s+to\s+[a-z]+", 

79 r"you\s+(?:need|have)\s+to\s+[a-z]+", 

80 r"kailangan\s+(?:mo|niyo|ninyo|nating|natin)\s+(?:ng\s+)?[a-z]+", 

81 r"(?:ipasa|isumite|ipadala|sagutan|tapusin|bayaran|pirmahan)\b", 

82] 

83_TASK_RE = re.compile(r"(?<![\w-])(?:" + "|".join(_TASK_TRIGGERS) + r")", re.IGNORECASE) 

84 

85# A task phrase ends at sentence punctuation, a line break, or a deadline connective. 

86_TASK_STOP_RE = re.compile( 

87 r"[.!?;\n]|,\s*(?:thanks|thank\s+you|salamat)\b|\s(?:by|before|until|no\s+later\s+than|on\s+or\s+before|hanggang|bago)\s", 

88 re.IGNORECASE, 

89) 

90_TRAILING_FILLER_RE = re.compile( 

91 r"(?:[\s,:\-]+|\s+(?:po|naman|lang|na|ha|please|asap|sa|on|ng|for|to|by|at|in|of|until|before)(?![\w-]))+$", 

92 re.IGNORECASE, 

93) 

94_DEADLINE_MARKER_RE = re.compile( 

95 r"\b(?:by|before|until|due(?:\s+(?:date|on|by))?|deadline|no\s+later\s+than|on\s+or\s+before|hanggang|bago|sa\s+or\s+bago|takdang\s+petsa)\b[\s:,]*(?:sa\s+|on\s+|the\s+)?$" 

96)

99def _find_dates(text: str) -> list[RawSpan]: 

100 return [RawSpan("DATE", match.group(0), match.start(), match.end()) for match in _DATE_RE.finditer(text)] 

101 

102 

103def _find_tasks(text: str, dates: list[RawSpan]) -> list[RawSpan]: 

104 spans: list[RawSpan] = [] 

105 cursor = 0 

106 for match in _TASK_RE.finditer(text): 

107 start = match.start() 

108 if start < cursor or any(date.start <= start < date.end for date in dates): 

109 continue 

110 limit = min(len(text), start + MAX_TASK_CHARS) 

111 stop = _TASK_STOP_RE.search(text, match.end(), limit) 

112 end = stop.start() if stop else limit 

113 for date in dates: 

114 if match.end() <= date.start < end: 

115 end = date.start 

116 break 

117 phrase = _TRAILING_FILLER_RE.sub("", text[start:end]) 

118 end = start + len(phrase) 

119 if end <= start: 

120 continue 

121 spans.append(RawSpan("TASK", phrase, start, end)) 

122 cursor = end 

123 return spans
