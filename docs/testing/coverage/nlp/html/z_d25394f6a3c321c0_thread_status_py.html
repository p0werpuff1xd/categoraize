1"""Local Reply Zero status from tagger signals (local first, LLM cascade). 

2 

3Statuses, from the user's point of view: 

4- TO_REPLY: someone is waiting on the user (a request or question to them, or 

5 the user promised something and has not delivered it yet). 

6- AWAITING_REPLY: the user asked or requested something and is waiting. 

7- FYI: nothing is asked of the user (never used when the user wrote last). 

8- ACTIONED: the exchange is done (thanks, confirmation, the user took 

9 ownership, or the user sent what was asked). 

10 

11The signals per message: a request (Bi-LSTM-CRF TASK span), a question 

12("?" or a Filipino question marker), a promise of a later reply or 

13deliverable, ownership ("I'll handle it"), and closing-only messages 

14("Thanks!", "Noted po"). Cases the rules cannot settle are marked not 

15confident, so the web app may ask the external LLM (only with consent). 

16""" 

17 

18import re 

19from collections.abc import Callable 

20from dataclasses import dataclass, field 

21 

22TO_REPLY, AWAITING_REPLY, FYI, ACTIONED = "TO_REPLY", "AWAITING_REPLY", "FYI", "ACTIONED" 

23 

24_QUESTION_RE = re.compile( 

25 r"\?|\b(?:pwede\s+(?:po\s+)?(?:ba|bang)|puwede\s+(?:po\s+)?(?:ba|bang)|kailan|saan|paano|ano\s+po|sino|" 

26 r"magkano|meron\s+(?:po\s+)?ba|may\s+(?:\w+\s+)?ba)\b", 

27 re.IGNORECASE, 

28) 

29_PROMISE_RE = re.compile( 

30 r"\b(?:i(?:'|’)?ll|i\s+will|we(?:'|’)?ll|we\s+will)\b[^.!?\n]{0,40}?\b(?:send|get\s+back|follow\s+up|update\s+you|" 

31 r"reply|share|forward|let\s+you\s+know|confirm|email\s+you)\b|" 

32 r"\b(?:babalikan\s+(?:kita|ko\s+kayo|ko)|ipapadala\s+ko|isesend\s+ko|i-?send\s+ko|ise-?send\s+ko|" 

33 r"sasabihan\s+(?:kita|ko\s+kayo)|ia-?update\s+(?:kita|ko\s+kayo)|magre-?reply\s+ako)\b", 

34 re.IGNORECASE, 

35) 

36_OWNERSHIP_RE = re.compile( 

37 r"\b(?:i(?:'|’)?ll|i\s+will|we(?:'|’)?ll)\s+(?:handle|take\s+care\s+of|fix|sort\s+out|deal\s+with)\b|" 

38 r"\b(?:ako\s+na\s+(?:bahala|ang\s+bahala)|aayusin\s+ko)\b|" 

39 r"\b(?:done|sent|attached|submitted|na-?send\s+ko\s+na|naipasa\s+ko\s+na|nai-?send\s+ko\s+na|tapos\s+na)\b", 

40 re.IGNORECASE, 

41) 

42_CLOSING_RE = re.compile( 

43 r"^(?:thanks?|thank\s+you(?:\s+so\s+much)?|ty|noted|ok(?:ay)?|got\s+it|received|sige|salamat|" 

44 r"maraming\s+salamat|noted\s+po|ok\s+po|sige\s+po|salamat\s+po|thanks\s+po|perfect|great|" 

45 r"awesome|appreciated|copy|nice)(?:[\s,.!]+(?:po|so\s+much|a\s+lot|again|ha|everyone|team|\w+))*[\s.!]*$", 

46 re.IGNORECASE, 

47)

75def message_signals(message: ThreadMessage, tag: Callable[[str], list]) -> Signals: 

76 text = f"{message.subject}\n{message.body}".strip() if not message.from_user else message.body.strip() 

77 body = message.body.strip() 

78 return Signals( 

79 request=any(span.label == "TASK" for span in tag(body)), 

80 question=bool(_QUESTION_RE.search(body)), 

81 promise=bool(_PROMISE_RE.search(text)), 

82 ownership=bool(_OWNERSHIP_RE.search(body)), 

83 closing=bool(_CLOSING_RE.match(_first_line(body))) and len(re.findall(r"\w+", body)) <= 10, 

84 ) 

85 

86 

87def asks_question(text: str) -> bool: 

88 return bool(_QUESTION_RE.search(text))

91def determine_status(messages: list[ThreadMessage], tag: Callable[[str], list]) -> StatusDecision: 

92 if not messages: 

93 return StatusDecision(FYI, False, "empty thread") 

94 last = messages[-1] 

95 signals = message_signals(last, tag) 

96 summary = signals.__dict__ 

97 

98 if last.from_user: 

99 if signals.promise: 

100 return StatusDecision(TO_REPLY, True, "you promised to send or reply later", summary) 

101 if signals.question: 

102 return StatusDecision(AWAITING_REPLY, True, "you asked something and are waiting", summary) 

103 # "I'll handle it" closes the request even though it reads like a task. 

104 if signals.ownership or signals.closing: 

105 return StatusDecision(ACTIONED, True, "you confirmed, thanked or took care of it", summary) 

106 if signals.request: 

107 return StatusDecision(AWAITING_REPLY, True, "you requested something and are waiting", summary) 

108 return StatusDecision(ACTIONED, False, "you replied without asking anything; unclear if done", summary) 

109 

110 if last.automated: 

111 return StatusDecision(FYI, True, "automated message", summary) 

112 # "I'll check and get back to you" puts the ball in their court, unless they 

113 # also asked the user something. 

114 if signals.promise and not signals.question: 

115 return StatusDecision(AWAITING_REPLY, True, "they promised to get back to you", summary) 

116 if signals.request or signals.question: 

117 return StatusDecision(TO_REPLY, True, "they asked you something", summary) 

118 if signals.closing: 

119 return StatusDecision(ACTIONED, True, "they thanked or confirmed", summary) 

120 

121 # Informational last message: an earlier request to the user that the user 

122 # never answered still counts. 

123 pending = _pending_request_before(messages, tag) 

124 if pending: 

125 return StatusDecision(TO_REPLY, False, "an earlier request to you looks unanswered", summary) 

126 return StatusDecision(FYI, len(messages) == 1, "nothing is asked of you", summary) 

127 

128 

129def _pending_request_before(messages: list[ThreadMessage], tag: Callable[[str], list]) -> bool: 

130 for message in reversed(messages[:-1]): 

131 if message.from_user: 

132 return False 

133 earlier = message_signals(message, tag) 

134 if earlier.request or earlier.question: 

135 return True 

136 return False

139def _first_line(text: str) -> str: 

140 return next((line.strip() for line in text.splitlines() if line.strip()), "")
